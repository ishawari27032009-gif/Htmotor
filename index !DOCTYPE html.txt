        function populateFilterOptions() {
            const plants = [...new Set(motorsData.map(d => d.Unit).filter(Boolean))];
            const sections = [...new Set(motorsData.map(d => d.Section).filter(Boolean))];

            const plantSel = document.getElementById('plantFilter');
            plantSel.innerHTML = '<option value="">All Plants</option>' + plants.map(p => `<option value="${p}">${p}</option>`).join('');

            const secSel = document.getElementById('sectionFilter');
            secSel.innerHTML = '<option value="">All Sections</option>' + sections.map(s => `<option value="${s}">${s}</option>`).join('');
        }

        function renderTable() {
            const searchStr = document.getElementById('searchInput').value.toLowerCase();
            const selectedPlant = document.getElementById('plantFilter').value;
            const selectedSection = document.getElementById('sectionFilter').value;
            const selectedCond = document.getElementById('conditionFilter').value;

            const filtered = motorsData.filter(item => {
                const matchSearch = (item.Tag || '').toLowerCase().includes(searchStr) ||
                                    (item.Desc || '').toLowerCase().includes(searchStr) ||
                                    (item.Make || '').toLowerCase().includes(searchStr) ||
                                    (item.Unit || '').toLowerCase().includes(searchStr);
                const matchPlant = !selectedPlant || item.Unit === selectedPlant;
                const matchSec = !selectedSection || item.Section === selectedSection;
                const matchCond = !selectedCond || (item.Condition || '').toLowerCase().includes(selectedCond.toLowerCase());
                return matchSearch && matchPlant && matchSec && matchCond;
            });

            // Update KPIs
            document.getElementById('statTotal').innerText = motorsData.length;
            document.getElementById('statReady').innerText = motorsData.filter(d => (d.Condition||'').toLowerCase().includes('ready')).length;
            document.getElementById('statOnOrder').innerText = motorsData.filter(d => (d.Condition||'').toLowerCase().includes('order')).length;
            document.getElementById('statPlants').innerText = [...new Set(motorsData.map(d => d.Unit))].length;

            const tbody = document.getElementById('motorTableBody');
            tbody.innerHTML = '';

            filtered.forEach((item, index) => {
                const globalIndex = motorsData.indexOf(item);
                const badgeColor = (item.Condition || '').toLowerCase().includes('ready') ? 'bg-emerald-100 text-emerald-800' : 'bg-amber-100 text-amber-800';

                const tr = document.createElement('tr');
                tr.className = 'hover:bg-slate-50 border-b border-slate-200';
                tr.innerHTML = `
                    <td class="p-3 font-semibold text-blue-900">${item.Unit || '-'}</td>
                    <td class="p-3">${item.Line || '-'}</td>
                    <td class="p-3">${item.Section || '-'}</td>
                    <td class="p-3 font-mono font-bold text-slate-900">${item.Tag || '-'}</td>
                    <td class="p-3">${item.Desc || '-'}</td>
                    <td class="p-3">${item.Make || '-'}</td>
                    <td class="p-3">${item.Type || '-'}</td>
                    <td class="p-3 font-semibold">${item.KW || '-'}</td>
                    <td class="p-3">${item.StatorVolt || '-'}</td>

        function closeModal() {
            document.getElementById('motorModal').classList.add('hidden');
        }

        function saveMotor(e) {
            e.preventDefault();
            const idx = parseInt(document.getElementById('editIndex').value);
            const newItem = {
                Unit: document.getElementById('inputUnit').value,
                Line: document.getElementById('inputLine').value,
                Section: document.getElementById('inputSection').value,
                Tag: document.getElementById('inputTag').value,
                Desc: document.getElementById('inputDesc').value,
                Make: document.getElementById('inputMake').value,
                Type: document.getElementById('inputType').value,
                KW: document.getElementById('inputPower').value,
                Frame: document.getElementById('inputFrame').value,
                StatorVolt: document.getElementById('inputStatorVolt').value,
                StatorCurr: document.getElementById('inputStatorCurr').value,
                RPM: document.getElementById('inputRPM').value,
                SpareAvail: document.getElementById('inputSpareAvail').value,
                Condition: document.getElementById('inputCondition').value,
                Remarks: document.getElementById('inputRemarks').value
            };

            if (idx === -1) {
                motorsData.push(newItem);
            } else {
                motorsData[idx] = newItem;
            }

            saveToStorage();
            populateFilterOptions();
            renderTable();
            closeModal();
        }

        function deleteMotor(index) {
            if(confirm("Are you sure you want to delete this motor entry?")) {
                motorsData.splice(index, 1);
                saveToStorage();
                renderTable();
            }
        }

        function exportCSV() {
            let csvContent = "data:text/csv;charset=utf-8,Plant,Line,Section,Tag,Description,Make,Type,Power_KW,Frame,StatorVolt,StatorCurr,RPM,SpareAvailable,Condition,Remarks\n";
            motorsData.forEach(d => {
                let row = [
                    `"${d.Unit || ''}"`,
                    `"${d.Line || ''}"`,
                    `"${d.Section || ''}"`,
                    `"${d.Tag || ''}"`,
                    `"${d.Desc || ''}"`,
                    `"${d.Make || ''}"`,
                    `"${d.Type || ''}"`,
                    `"${d.KW || ''}"`,
                    `"${d.Frame || ''}"`,
                    `"${d.StatorVolt || ''}"`,
                    `"${d.StatorCurr || ''}"`,
                    `"${d.RPM || ''}"`,
                    `"${d.SpareAvail || ''}"`,
                    `"${d.Condition || ''}"`,
                    `"${d.Remarks || ''}"`
                ].join(",");
                csvContent += row + "\n";
            });
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", "Critical_HT_Motors_MultiPlant.csv");
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
